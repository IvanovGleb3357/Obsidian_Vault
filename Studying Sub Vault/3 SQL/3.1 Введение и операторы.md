## Выбор определенных столбцов

	SELECT
	    id
	    manager name AS name
	FROM 
	    sales

## Фильтр вывода

	SELECE *
	FROM sales
	WHERE manager = 'dasd'
	OR manager = 'sasa'

## Сортировка вывода

	OREDER BY
		value DESC

	LIMIT 3
